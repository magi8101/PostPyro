PostPyro

Async PostgreSQL Driver for Python, Built with Rust

Built with PyO3 and sqlx, PostPyro is a fully async driver - every I/O method is async def and releases the GIL while waiting on Postgres - with DB-API 2.0-flavored exception handling.

quick_example.py 
import asyncio
import PostPyro

async def main():
    # Connect to PostgreSQL (returns a Pool)
    pool = await PostPyro.connect("postgresql://user:pass@localhost/db")

    # Query data
    users = await pool.query("SELECT * FROM users WHERE active = $1", [True])
    for user in users:
        print(f"User: {user['name']} ({user['email']})")

    # Execute updates
    affected = await pool.execute(
        "UPDATE users SET last_login = NOW() WHERE id = $1",
        [user_id]
    )

    # Use transactions
    tx = await pool.transaction()
    async with tx:
        await tx.execute("INSERT INTO orders (user_id) VALUES ($1)", [user_id])
        await tx.execute("UPDATE inventory SET stock = stock - 1")

    await pool.close()

asyncio.run(main())

Key Features

🔥

Rust Core

The driver itself is a PyO3 extension module - query execution and type conversion happen in compiled Rust, not interpreted Python.

🛡️

Memory Safe

Rust's ownership system prevents memory leaks and segfaults.

🌐

Broad Type Support

Booleans, integers, floats, numeric, text, dates/times, UUIDs, and JSON/JSONB convert automatically. See the Type System reference below for the full list.

⚡

Fully Async

Every I/O method is async def, built on sqlx's connection pooling and Tokio runtime, and releases the GIL while waiting on Postgres.

🔒

Type Conversion

Postgres values decode to native Python types automatically. Bound parameters other than bool/int/float/str/None need an explicit ::type cast in the SQL text - see the Type System reference below.

🎯

DB-API 2.0 Compliant

Standard Python database interface ensures compatibility with existing code.

Quick Start

import asyncio
import PostPyro

async def main():
    # Connect to database (returns a Pool)
    pool = await PostPyro.connect("postgresql://user:password@localhost:5432/mydb")

    # Create a table
    await pool.execute("""
        CREATE TABLE users (
            id SERIAL PRIMARY KEY,
            name TEXT NOT NULL,
            email TEXT UNIQUE,
            age INTEGER
        )
    """)

    # Insert data
    affected = await pool.execute(
        "INSERT INTO users (name, email, age) VALUES ($1, $2, $3)",
        ["Alice", "alice@example.com", 30]
    )
    print(f"Inserted {affected} rows")

    # Close the pool
    await pool.close()

asyncio.run(main())
# Query multiple rows
users = await pool.query("SELECT * FROM users WHERE age > $1", [25])
for user in users:
    print(f"ID: {user['id']}, Name: {user['name']}, Age: {user['age']}")

# Query single row (raises ProgrammingError on zero matches;
# extra matches beyond the first are silently discarded)
user = await pool.query_one("SELECT * FROM users WHERE id = $1", [1])
print(f"Found user: {user['name']}")

# There's no execute_batch() - run each statement individually
names_and_ages = [("Bob", 25), ("Charlie", 35), ("Diana", 28)]
inserted = 0
for name, age in names_and_ages:
    inserted += await pool.execute(
        "INSERT INTO users (name, age) VALUES ($1, $2)", [name, age]
    )
print(f"Inserted {inserted} rows")

# Pool status check
if not pool.is_closed():
    print("✅ Pool is open")
# pool.transaction() is a real BEGIN round-trip, so it must be
# awaited before it can be used as a context manager -
# `async with pool.transaction():` raises TypeError.
tx = await pool.transaction()
async with tx:
    await tx.execute("INSERT INTO users (name) VALUES ($1)", ["Alice"])
    await tx.execute("UPDATE accounts SET balance = balance + 100 WHERE user_id = $1", [1])
    # Automatically commits on success, rolls back on exception

# Manual transaction management (commit()/rollback() also
# work outside a `with` block)
tx = await pool.transaction()
try:
    await tx.execute("INSERT INTO orders (user_id, amount) VALUES ($1, $2)", [1, 99.99])
    await tx.execute("UPDATE inventory SET stock = stock - 1 WHERE product_id = $1", [123])
    await tx.commit()
except Exception as e:
    await tx.rollback()
    print(f"Transaction failed: {e}")

# Query within a transaction
tx = await pool.transaction()
async with tx:
    users = await tx.query("SELECT * FROM users WHERE created_today = true")
    for user in users:
        await tx.execute("UPDATE users SET welcomed = true WHERE id = $1", [user['id']])
# bind_params only auto-converts bool/int/float/str/None -
# anything else (date, datetime, uuid, dict/list for JSON) has
# to be passed as a string with an explicit Postgres cast.
import json
import uuid
from datetime import date, datetime

await pool.execute("""
    INSERT INTO complex_data (
        json_col, uuid_col, date_col, timestamp_col
    ) VALUES ($1::jsonb, $2::uuid, $3::date, $4::timestamp)
""", [
    json.dumps({"name": "John", "scores": [85, 92, 78]}),  # JSON
    str(uuid.uuid4()),                                      # UUID
    date.today().isoformat(),                               # Date
    datetime.now().isoformat(),                             # Timestamp
])

# Reading back decodes to the right Python type automatically -
# this direction IS automatic (see the Type System reference below).

# Convert Row to dict for pandas integration
import pandas as pd
rows = await pool.query("SELECT * FROM users")
df = pd.DataFrame([row.to_dict() for row in rows])

Module Functions

await PostPyro.connect(dsn, max_size=10, min_size=0)

Create a connection pool. This is the only way to get a Pool - there's no synchronous constructor, since establishing the first connection is itself async.

Parameters:
  • dsn (str): PostgreSQL connection string
  • max_size (int, optional): maximum pool connections (default 10)
  • min_size (int, optional): minimum pool connections (default 0)
Returns:

Pool object

Example:
pool = await PostPyro.connect("postgresql://user:pass@localhost:5432/mydb", max_size=20)
PostPyro.__version__ / .apilevel / .threadsafety / .paramstyle

Module-level constants. No get_version() function - read __version__ directly.

Example:
print(PostPyro.__version__)   # driver version
print(PostPyro.apilevel)      # "2.0" - DB-API 2.0-flavored
print(PostPyro.paramstyle)    # "numeric" - uses $1, $2, ... parameters

Pool Class

Obtained via await PostPyro.connect(...). Pool and the old ConnectionPool class were unified into this one type - there's no separate Connection class to construct.

await pool.execute(query, params=None)

Execute INSERT, UPDATE, DELETE, or DDL statements.

Parameters:
  • query (str): SQL query string
  • params (list, optional): Query parameters using $1, $2, ... placeholders
Returns:

Number of rows affected (int)

Examples:
# INSERT
affected = await pool.execute("INSERT INTO users (name, age) VALUES ($1, $2)", ["Alice", 30])

# UPDATE
affected = await pool.execute("UPDATE users SET age = $1 WHERE name = $2", [31, "Alice"])

# DELETE
affected = await pool.execute("DELETE FROM users WHERE age < $1", [18])
await pool.query(query, params=None)

Execute SELECT queries and return all matching rows.

Parameters:
  • query (str): SQL SELECT statement
  • params (list, optional): Query parameters
Returns:

List of Row objects

Example:
rows = await pool.query("SELECT id, name, age FROM users WHERE age > $1", [25])
for row in rows:
    print(f"ID: {row['id']}, Name: {row['name']}, Age: {row['age']}")
await pool.query_one(query, params=None)

Execute a SELECT and return its first row.

Parameters:
  • query (str): SQL SELECT statement
  • params (list, optional): Query parameters
Returns:

Single Row object

Raises:

ProgrammingError if zero rows are returned. If more than one row matches, the extras are silently discarded rather than raising - only use this when your query itself guarantees at most one row (a primary-key lookup, LIMIT 1, etc.).

Example:
user = await pool.query_one("SELECT * FROM users WHERE id = $1", [1])
print(f"User name: {user['name']}")
await pool.transaction()

Start a new transaction (a real BEGIN). Await it, then use the result as an async with block - async with pool.transaction(): raises TypeError because the un-awaited coroutine has no __aenter__.

Returns:

Transaction object (async context manager)

Example:
tx = await pool.transaction()
async with tx:
    await tx.execute("INSERT INTO users (name) VALUES ($1)", ["Alice"])
    await tx.execute("UPDATE accounts SET balance = balance - 100 WHERE user_id = $1", [1])
    # Automatically commits on success, rolls back on exception
await pool.close()

Close the pool and free its connections.

Example:
await pool.close()
pool.is_closed()

Check whether the pool has been closed. Synchronous - no await.

Returns:

bool

Example:
if not pool.is_closed():
    await pool.query("SELECT 1")

Row Class

Represents a single row from a query result with dict-like interface.

row[key]

Access column values by index or name. Negative indices (row[-1]) work like a Python list.

Example:
row = await pool.query_one("SELECT id, name, email FROM users WHERE id = $1", [1])

# Access by column name
print(row['name'])
print(row['email'])

# Access by index
print(row[0])   # id
print(row[1])   # name
print(row[-1])  # last column
row.to_dict()

Convert row to a Python dictionary.

Returns:

Dictionary with column names as keys

Example:
user_dict = row.to_dict()
print(user_dict)  # {'id': 1, 'name': 'Alice', 'email': 'alice@example.com'}

Transaction Class

Obtained via await pool.transaction(). Using a transaction after it has been committed or rolled back raises ProgrammingError rather than hanging or panicking.

await tx.execute(query, params=None)

Execute a statement within the transaction.

Example:
tx = await pool.transaction()
async with tx:
    await tx.execute("INSERT INTO users (name) VALUES ($1)", ["Alice"])
    await tx.execute("UPDATE accounts SET balance = balance - $1 WHERE id = $2", [100, 1])
await tx.query(query, params=None) / await tx.query_one(query, params=None)

Execute a query within the transaction. Same semantics as pool.query/pool.query_one.

Example:
tx = await pool.transaction()
async with tx:
    users = await tx.query("SELECT * FROM users WHERE created_today = true")
    for user in users:
        await tx.execute("UPDATE users SET welcomed = true WHERE id = $1", [user['id']])
await tx.commit() / await tx.rollback() / tx.is_active()

Explicit commit/rollback - also work outside a with block. is_active() is synchronous and returns False after commit/rollback.

Example:
tx = await pool.transaction()
try:
    await tx.execute("INSERT INTO orders (user_id) VALUES ($1)", [1])
    await tx.commit()
except Exception:
    await tx.rollback()
print(tx.is_active())  # False

Error Handling

PostPyro maps sqlx/Postgres errors onto a DB-API 2.0-flavored exception hierarchy, so you can catch by class instead of parsing SQLSTATE codes yourself.

Exception Hierarchy

DatabaseError                    # Base database error
├── InterfaceError              # Driver interface problems
├── DataError                   # Data processing errors
├── OperationalError            # Database operation errors
├── IntegrityError              # Constraint violations
├── InternalError               # Internal database errors
├── ProgrammingError            # SQL programming errors
└── NotSupportedError           # Unsupported operations
Error Handling Example:
import PostPyro

async def main():
    pool = await PostPyro.connect("postgresql://user:pass@localhost/db")
    try:
        await pool.execute("INSERT INTO users (email) VALUES ($1)", ["invalid-email"])

    except PostPyro.IntegrityError as e:
        print(f"Constraint violation: {e}")

    except PostPyro.OperationalError as e:
        print(f"Database operation failed: {e}")

    except PostPyro.ProgrammingError as e:
        print(f"SQL syntax error: {e}")

    except PostPyro.DatabaseError as e:
        print(f"General database error: {e}")

Type System

PostPyro automatically converts between Python and PostgreSQL types.

PostgreSQL Type Python Type Example
BOOLEAN bool True, False
SMALLINT, INTEGER, BIGINT int 42, -123, 9223372036854775807
REAL, DOUBLE PRECISION float 3.14, 2.718
NUMERIC decimal.Decimal Decimal('1234.5678')
TEXT, VARCHAR, CHAR, NAME str "Hello World"
DATE datetime.date date(2023, 12, 25)
TIME datetime.time time(14, 30, 0)
TIMESTAMP datetime.datetime datetime(2023, 12, 25, 14, 30)
TIMESTAMPTZ datetime.datetime With timezone info (UTC)
UUID str '550e8400-e29b-...'
JSON, JSONB dict, list, etc. {"key": "value"}, [1, 2, 3]

Any other type (BYTEA, arrays, INET/CIDR, custom/enum types, etc.) isn't decodable yet - reading such a column raises PostPyro.NotSupportedError naming the type, rather than silently returning the wrong value or None.

🚀 FastAPI Integration

from fastapi import FastAPI, HTTPException
import PostPyro

app = FastAPI()

@app.on_event("startup")
async def startup():
    app.state.pool = await PostPyro.connect("postgresql://user:pass@localhost/db")

@app.get("/users/{user_id}")
async def get_user(user_id: int):
    try:
        user = await app.state.pool.query_one(
            "SELECT id, name, email FROM users WHERE id = $1",
            [user_id]
        )
        return user.to_dict()
    except PostPyro.DatabaseError:
        raise HTTPException(status_code=404, detail="User not found")

🔧 Pooling Is Built In

import asyncio
import PostPyro

async def main():
    # Pool is already a connection pool - no wrapper class needed.
    # max_size/min_size configure how many connections it keeps.
    pool = await PostPyro.connect(
        "postgresql://user:pass@localhost/db",
        max_size=20,
        min_size=2,
    )

    # Concurrent queries share the pool automatically
    results = await asyncio.gather(
        pool.query("SELECT * FROM users WHERE id = $1", [1]),
        pool.query("SELECT * FROM users WHERE id = $1", [2]),
    )

    if not pool.is_closed():
        await pool.close()

asyncio.run(main())

💾 Batch Processing

async def bulk_insert_users(pool, users_data):
    # Single transaction - all rows commit or roll back together
    tx = await pool.transaction()
    async with tx:
        for user_data in users_data:
            await tx.execute(
                "INSERT INTO users (name, email, age) VALUES ($1, $2, $3)",
                [user_data['name'], user_data['email'], user_data['age']]
            )

    # There's no execute_batch() - looping awaited execute() calls
    # is the working pattern (still parameterized, no string formatting)
    for user_data in users_data:
        await pool.execute(
            "INSERT INTO users (name, email, age) VALUES ($1, $2, $3)",
            [user_data['name'], user_data['email'], user_data['age']]
        )

Performance Benchmark

import asyncio
import time
import PostPyro

async def main():
    pool = await PostPyro.connect("postgresql://user:pass@localhost/db")

    # Benchmark: Insert 10,000 records concurrently across the pool
    start_time = time.time()

    await asyncio.gather(*(
        pool.execute("INSERT INTO benchmark (value) VALUES ($1)", [i])
        for i in range(10000)
    ))

    elapsed = time.time() - start_time
    print(f"Inserted 10,000 records in {elapsed:.2f} seconds")
    print(f"Rate: {10000/elapsed:.0f} inserts/second")

    await pool.close()

asyncio.run(main())
